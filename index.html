/**
 * 彰基智慧醫療大挑戰 - Google Apps Script 後端 v3
 *
 * 功能：
 *  1. 題庫 / 設定同步（Google 試算表：Questions、Config、Logs 三個工作表，自動建立）
 *  2. 素材自動上傳：後台上傳的圖片、MP3 會自動存進 Google Drive 指定資料夾
 *  3. 素材代理讀取：網頁透過本程式讀取 Drive 檔案（不再受「連結直連被 Google 擋住」影響）
 *  4. 連線診斷（後台「診斷連線」按鈕使用）
 *
 * 【第一次使用】編輯器上方函式選單選「setup」按「執行」，依畫面完成授權。
 * 【部署】網頁應用程式以「我」的身分執行；存取權設為「所有人」。
 * 【修改程式後】部署 → 管理部署作業 → 編輯(鉛筆) → 版本選「新版本」→ 部署。
 */

// ★ 選填：貼上你自己建立的 Drive 資料夾 ID（資料夾網址 .../folders/ 後面那一串）。
//   留空 = 自動在「我的雲端硬碟」建立名為下方 FOLDER_NAME 的資料夾。
const FOLDER_ID = '1q8Y_nNdExN9ACUehnkZePazBGneKlX8d';
const FOLDER_NAME = '彰基智慧醫療大挑戰_素材';

const VERSION = 'v3.0';
const MAX_FILE_BYTES = 20 * 1024 * 1024;
const Q_HEADERS = ['ID', '題目', 'A選項', 'B選項', 'C選項', '正確答案'];
const C_HEADERS = ['Key', 'Value'];
const L_HEADERS = ['時間', '總分', '答對題數', '挑戰結果'];

/* ------------------------------------------------------------------ 入口 */

function setup() {
  const ss = getSS_();
  ensureSheets_(ss);
  const folder = getFolder_();
  Logger.log('試算表：' + ss.getUrl());
  Logger.log('素材資料夾：' + folder.getUrl());
}

function doGet(e) {
  try {
    const p = (e && e.parameter) || {};
    if (p.action === 'bridge') return bridgePage_(p);
    if (p.action === 'ping') return json_(pingInfo_());
    if (p.action === 'file') return json_(readFile_(p.id));
    return json_(readAll_());
  } catch (err) {
    return json_({ status: 'error', message: errMsg_(err) });
  }
}

function doPost(e) {
  let lock = null;
  try {
    const payload = JSON.parse((e && e.postData && e.postData.contents) || '{}');
    const action = payload.action;

    if (action === 'PING') {
      PropertiesService.getScriptProperties().setProperty('LAST_PING', new Date().toISOString());
      return json_({ status: 'ok', version: VERSION, time: new Date().toISOString() });
    }

    if (action === 'UPLOAD_ASSET') {
      return json_(uploadAsset_(payload));
    }

    if (action === 'CHECK_DRIVE_WRITE') {
      const folder = getFolder_();
      const testFile = folder.createFile('upload_check_' + Date.now() + '.txt', 'write-check', MimeType.PLAIN_TEXT);
      const id = testFile.getId();
      testFile.setTrashed(true);
      return json_({ status: 'ok', folderUrl: folder.getUrl(), testFileId: id });
    }

    lock = LockService.getScriptLock();
    lock.waitLock(20000);
    const ss = getSS_();
    ensureSheets_(ss);

    if (action === 'SYNC_UP') {
      let qCount = null;
      let cCount = null;
      if (Array.isArray(payload.questions)) {
        const rows = payload.questions.map(function (q) {
          return [str_(q.id), str_(q.question), str_(q.optionA), str_(q.optionB), str_(q.optionC), str_(q.answer)];
        });
        writeTable_(ss.getSheetByName('Questions'), Q_HEADERS, rows);
        qCount = rows.length;
      }
      if (payload.config && typeof payload.config === 'object') {
        // 與現有設定合併（保留你在試算表手動加的欄位）
        const merged = readConfig_(ss);
        Object.keys(payload.config).forEach(function (k) {
          const v = payload.config[k];
          const s = (v !== null && typeof v === 'object') ? JSON.stringify(v) : str_(v);
          if (s.length < 49000) merged[k] = s;
        });
        const rows = Object.keys(merged).map(function (k) { return [k, merged[k]]; });
        writeTable_(ss.getSheetByName('Config'), C_HEADERS, rows);
        cCount = rows.length;
      }
      return json_({ status: 'ok', questions: qCount, config: cCount, sheetUrl: ss.getUrl() });
    }

    if (action === 'LOG_RESULT') {
      ss.getSheetByName('Logs').appendRow([new Date(), payload.score, payload.correctCount, payload.outcome]);
      return json_({ status: 'ok' });
    }

    return json_({ status: 'error', message: '未知的 action：' + action });
  } catch (err) {
    return json_({ status: 'error', message: errMsg_(err) });
  } finally {
    if (lock) { try { lock.releaseLock(); } catch (x) {} }
  }
}

/* GitHub Pages 跨網域通訊：GAS HTML Service 在內嵌頁用 google.script.run
   取得真正的回應，再用 postMessage 回傳。不可把任意網頁傳來的 token 當密鑰。 */
function bridgePage_(p) {
  const origin = String(p.origin || '');
  const channel = String(p.channel || '');
  if (!/^https:\/\/[^\s/?#]+(?::\d+)?$/.test(origin) || !/^[a-zA-Z0-9_-]{16,100}$/.test(channel)) {
    throw new Error('橋接來源或通道格式不正確');
  }
  const safe = function (v) { return JSON.stringify(v).replace(/</g, '\\u003c'); };
  const html = '<!doctype html><html><head><meta charset="UTF-8"></head><body>' +
    '<script>"use strict";const origin=' + safe(origin) + ',channel=' + safe(channel) + ';' +
    'function send(value){window.top.postMessage(Object.assign({smcBridge:channel},value),origin)}' +
    'window.addEventListener("message",function(e){' +
    'if(e.source!==window.top||e.origin!==origin||!e.data||e.data.smcBridge!==channel||e.data.type!=="request")return;' +
    'var id=e.data.id;' +
    'google.script.run.withSuccessHandler(function(r){send({type:"response",id:id,result:r})})' +
    '.withFailureHandler(function(err){send({type:"response",id:id,error:String(err&&err.message||err)})})' +
    '.bridgeRequest(e.data.request);' +
    '});send({type:"ready"});<\/script></body></html>';
  return HtmlService.createHtmlOutput(html).setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

function bridgeRequest(req) {
  if (!req || typeof req !== 'object') throw new Error('要求格式錯誤');
  if (req.method === 'GET') {
    const allowed = { ping: true, file: true, all: true };
    if (!allowed[req.action]) throw new Error('不支援的讀取操作');
    const output = doGet({ parameter: { action: req.action, id: req.id || '' } });
    return JSON.parse(output.getContent());
  }
  if (req.method === 'POST') {
    const allowed = { PING: true, UPLOAD_ASSET: true, SYNC_UP: true, LOG_RESULT: true, CHECK_DRIVE_WRITE: true };
    if (!allowed[req.payload && req.payload.action]) throw new Error('不支援的寫入操作');
    const output = doPost({ postData: { contents: JSON.stringify(req.payload) } });
    return JSON.parse(output.getContent());
  }
  throw new Error('不支援的要求');
}

/* ------------------------------------------------------------------ 讀取 */

function readAll_() {
  const ss = getSS_();
  ensureSheets_(ss);
  const qRows = ss.getSheetByName('Questions').getDataRange().getValues();
  const questions = [];
  for (let i = 1; i < qRows.length; i++) {
    if (!qRows[i][1]) continue;
    questions.push({
      id: String(qRows[i][0] || 'q_' + i),
      question: String(qRows[i][1]),
      optionA: String(qRows[i][2]),
      optionB: String(qRows[i][3]),
      optionC: String(qRows[i][4]),
      answer: String(qRows[i][5]).trim().toUpperCase()
    });
  }
  return { status: 'ok', version: VERSION, questions: questions, config: readConfig_(ss) };
}

function readConfig_(ss) {
  const rows = ss.getSheetByName('Config').getDataRange().getValues();
  const cfg = {};
  for (let i = 1; i < rows.length; i++) {
    if (rows[i][0]) cfg[String(rows[i][0])] = String(rows[i][1] === null || rows[i][1] === undefined ? '' : rows[i][1]);
  }
  return cfg;
}

function pingInfo_() {
  const ss = getSS_();
  ensureSheets_(ss);
  const info = {
    status: 'ok',
    version: VERSION,
    sheetName: ss.getName(),
    sheetUrl: ss.getUrl(),
    questionRows: Math.max(0, ss.getSheetByName('Questions').getLastRow() - 1),
    configRows: Math.max(0, ss.getSheetByName('Config').getLastRow() - 1),
    logRows: Math.max(0, ss.getSheetByName('Logs').getLastRow() - 1),
    lastPing: PropertiesService.getScriptProperties().getProperty('LAST_PING') || ''
  };
  try {
    const folder = getFolder_();
    info.folderName = folder.getName();
    info.folderUrl = folder.getUrl();
    let n = 0;
    const it = folder.getFiles();
    while (it.hasNext() && n < 500) { it.next(); n++; }
    info.folderFiles = n;
  } catch (err) {
    info.driveError = errMsg_(err);
  }
  return info;
}

/* ------------------------------------------------------------------ Drive 素材 */

function uploadAsset_(payload) {
  const mime = String(payload.mime || '');
  if (!/^(image|audio)\//.test(mime)) throw new Error('只支援圖片或音訊檔案（收到：' + mime + '）');
  if (!payload.data) throw new Error('沒有收到檔案內容');

  const bytes = Utilities.base64Decode(payload.data);
  if (bytes.length > MAX_FILE_BYTES) throw new Error('檔案超過 20MB 上限');

  const slot = String(payload.slot || 'asset').replace(/[^\w-]/g, '');
  const safeName = String(payload.name || 'file').replace(/[\\/:*?"<>|]/g, '_');
  const stamp = Utilities.formatDate(new Date(), 'Asia/Taipei', 'yyyyMMdd-HHmmss');
  const blob = Utilities.newBlob(bytes, mime, slot + '_' + stamp + '_' + safeName);

  const folder = getFolder_();
  const file = folder.createFile(blob);
  if (!isAllowedFile_(file)) throw new Error('建立檔案後未能驗證素材資料夾');
  let shared = true;
  try {
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
  } catch (err) {
    shared = false; // 組織政策不允許公開時略過；網頁仍可透過本程式讀取
  }
  return {
    status: 'ok',
    id: file.getId(),
    name: file.getName(),
    size: bytes.length,
    shared: shared,
    url: file.getUrl(),
    folderUrl: folder.getUrl()
  };
}

function readFile_(id) {
  if (!id) throw new Error('缺少檔案 ID');
  const file = DriveApp.getFileById(id);
  if (!isAllowedFile_(file)) {
    throw new Error('此檔案不在素材資料夾內，且未開放「知道連結的任何人」，基於安全無法讀取');
  }
  if (file.getSize() > MAX_FILE_BYTES) throw new Error('檔案超過 20MB 上限');
  const blob = file.getBlob();
  return {
    status: 'ok',
    id: id,
    name: file.getName(),
    mime: fixMime_(blob.getContentType(), file.getName()),
    data: Utilities.base64Encode(blob.getBytes())
  };
}

function isAllowedFile_(file) {
  try {
    const acc = file.getSharingAccess();
    if (acc === DriveApp.Access.ANYONE_WITH_LINK || acc === DriveApp.Access.ANYONE) return true;
  } catch (err) {}
  const folderId = getFolder_().getId();
  const parents = file.getParents();
  while (parents.hasNext()) {
    if (parents.next().getId() === folderId) return true;
  }
  return false;
}

function fixMime_(mime, name) {
  if (mime && mime !== 'application/octet-stream') return mime;
  const ext = String(name).split('.').pop().toLowerCase();
  const map = { mp3: 'audio/mpeg', m4a: 'audio/mp4', wav: 'audio/wav', ogg: 'audio/ogg', aac: 'audio/aac',
                png: 'image/png', jpg: 'image/jpeg', jpeg: 'image/jpeg', webp: 'image/webp', gif: 'image/gif', svg: 'image/svg+xml' };
  return map[ext] || mime || 'application/octet-stream';
}

/* ------------------------------------------------------------------ 工具 */

function getSS_() {
  // 從試算表「擴充功能 → Apps Script」開啟的腳本：直接用綁定的試算表
  let ss = SpreadsheetApp.getActiveSpreadsheet();
  if (ss) return ss;
  // 獨立腳本（沒綁定試算表）：自動建立一份並記住
  const props = PropertiesService.getScriptProperties();
  const id = props.getProperty('SHEET_ID');
  if (id) {
    try { return SpreadsheetApp.openById(id); } catch (err) {}
  }
  ss = SpreadsheetApp.create('彰基智慧醫療大挑戰資料庫');
  props.setProperty('SHEET_ID', ss.getId());
  return ss;
}

function getFolder_() {
  if (FOLDER_ID) {
    try { return DriveApp.getFolderById(FOLDER_ID); }
    catch (err) { throw new Error('找不到 FOLDER_ID 指定的資料夾，請確認 ID 正確且此帳號有權限：' + FOLDER_ID); }
  }
  const props = PropertiesService.getScriptProperties();
  const saved = props.getProperty('FOLDER_ID');
  if (saved) {
    try { return DriveApp.getFolderById(saved); } catch (err) {}
  }
  const it = DriveApp.getFoldersByName(FOLDER_NAME);
  const folder = it.hasNext() ? it.next() : DriveApp.createFolder(FOLDER_NAME);
  props.setProperty('FOLDER_ID', folder.getId());
  return folder;
}

function ensureSheets_(ss) {
  getSheet_(ss, 'Questions', Q_HEADERS);
  getSheet_(ss, 'Config', C_HEADERS);
  getSheet_(ss, 'Logs', L_HEADERS);
  const blank = ss.getSheetByName('Sheet1') || ss.getSheetByName('工作表1');
  if (blank && ss.getSheets().length > 1 && blank.getLastRow() === 0) ss.deleteSheet(blank);
}

function getSheet_(ss, name, headers) {
  let sheet = ss.getSheetByName(name);
  if (!sheet) {
    sheet = ss.insertSheet(name);
    sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
    sheet.setFrozenRows(1);
  }
  return sheet;
}

// 整張表一次寫入，並設為純文字格式（避免 = + 開頭的題目被當成公式）
function writeTable_(sheet, headers, rows) {
  sheet.clearContents();
  const all = [headers].concat(rows);
  const range = sheet.getRange(1, 1, all.length, headers.length);
  range.setNumberFormat('@');
  range.setValues(all);
  sheet.setFrozenRows(1);
}

function str_(v) { return (v === null || v === undefined) ? '' : String(v); }
function errMsg_(err) { return String((err && err.message) || err); }

function json_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
